{{- /* Project override of Hextra's render-link hook. Adds resolution of RELATIVE *.md cross-references (e.g. `lazy-representation.md`) so they point at the sibling page URL (/spec/draft/lazy-representation/) instead of being resolved by the browser against the current page (/spec/draft/seq-representation/lazy-representation.md). Everything else is unchanged from the upstream hook. */ -}} {{- $dest := .Destination -}} {{- $url := urls.Parse $dest -}} {{- if and $dest (hasPrefix $dest "/") -}} {{- with or (.PageInner.GetPage $url.Path) (.PageInner.Resources.Get $url.Path) (resources.Get $url.Path) -}} {{- $query := cond $url.RawQuery (printf "?%s" $url.RawQuery) "" -}} {{- $fragment := cond $url.Fragment (printf "#%s" $url.Fragment) "" -}} {{- $dest = printf "%s%s%s" .RelPermalink $query $fragment -}} {{- else -}} {{- $dest = (relURL (strings.TrimPrefix "/" $dest)) -}} {{- end -}} {{- else if and $url.Path (not $url.IsAbs) (strings.HasSuffix $url.Path ".md") -}} {{- $fragment := cond $url.Fragment (printf "#%s" $url.Fragment) "" -}} {{- $base := strings.TrimSuffix ".md" (path.Base $url.Path) -}} {{- $target := or (.PageInner.GetPage $url.Path) (.PageInner.GetPage $base) -}} {{- if $target -}} {{- $dest = printf "%s%s" $target.RelPermalink $fragment -}} {{- else -}} {{- with .PageInner.Parent -}} {{- $dest = printf "%s%s/%s" .RelPermalink $base $fragment -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Bare-§ cross-references: spec source uses [§](chapter.md#anchor) where build.fsx (the MkDocs spec build) fills in the section number. Hugo doesn't run build.fsx, so the link renders as a meaningless naked "§". When the text is a bare § and there is an anchor, derive a readable section name from the anchor (kebab → Title Case) so it reads "§ Named Patterns" instead. */ -}} {{- $linkText := .Text -}} {{- if and (eq (strings.TrimSpace .Text) "§") $url.Fragment -}} {{- $name := replace $url.Fragment "-" " " | strings.Title -}} {{- $linkText = printf "§ %s" $name -}} {{- end -}} {{- with . -}} {{- $isExternal := strings.HasPrefix .Destination "http" -}} {{- $linkText | safeHTML -}} {{- if and .Page.Site.Params.externalLinkDecoration $isExternal -}} {{- partial "utils/icon.html" (dict "name" "arrow-up-right" "attributes" `class="hx:inline hx:rtl:rotate-270 hx:align-baseline" height="1em"`) -}} {{- end -}} {{- end -}}
